@[expose] def KakeyaSharp.SharpTubeBound : ℕ → Prop :=
fun n =>
  ∃ C,
    0 < C ∧
      ∀ (m : ℕ) (δ : ℝ),
        0 < δ →
          δ ≤ 1 →
            ∀ (v x : Fin m → KakeyaSharp.Space n),
              (∀ (i : Fin m), ‖v i‖ = 1) →
                (∀ (i j : Fin m), i ≠ j → δ ≤ dist (v i) (v j)) →
                  ∀ (a : Fin m → ℝ),
                    (∀ (i : Fin m), 0 ≤ a i) →
                      ∫ (y : KakeyaSharp.Space n),
                          (∑ i, (KakeyaSharp.tube (2 * δ) (v i) (x i)).indicator (fun x => a i) y) ^
                            KakeyaSharp.overlapExponent n ≤
                        C * Real.log (2 / δ) * δ ^ (n - 1) * ∑ i, a i ^ KakeyaSharp.overlapExponent n

  ∃ C,
